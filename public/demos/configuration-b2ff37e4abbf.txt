Complete the task

apiVersion: v1
kind: Pod
metadata:
  name: nginx
  labels:
    run: nginx
  namespace: limitrange
spec:
  containers:
    - name: nginx
      image: nginx:1.28
      imagePullPolicy: IfNotPresent
      resources:
        requests:
          memory: 250Mi
        limits:
          memory: 400Mi
KUBEQUEST_YAML
kubectl wait -n limitrange --for=condition=Ready pod/nginx --timeout=90s

Why the repair works
